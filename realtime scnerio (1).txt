public class CustomerValidation {

    public static List<String> validate(CustomerDto dto) {
        List<String> errors = new ArrayList<>();

        if (dto.getName() == null || dto.getName().trim().isEmpty()) {
            errors.add("name is required");
        }
        if (dto.getPhone() == null || !dto.getPhone().matches("^[6-9]\\d{9}$")) {
            errors.add("phone must be valid 10 digit number");
        }
        if (dto.getAddress() == null || dto.getAddress().trim().length() < 5) {
            errors.add("address must be at least 5 characters");
        }
        return errors;
    }
}

package com.iex.logistics.driverassignment.filter;

import com.fasterxml.jackson.databind.ObjectMapper;
import com.iex.logistics.driverassignment.dto.CustomerDto;
import com.iex.logistics.driverassignment.entity.EncryptedCustomerPayload;
import com.iex.logistics.driverassignment.repository.EncryptedCustomerPayloadRepository;
import com.iex.logistics.util.AesGcmUtil;
import jakarta.servlet.*;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;
import org.springframework.util.StreamUtils;

import javax.crypto.SecretKey;
import java.io.IOException;
import java.io.PrintWriter;
import java.nio.charset.StandardCharsets;
import java.time.LocalDateTime;
import java.util.List;

@Component
@Order(1)
public class DecryptFilter implements Filter {

    @Value("${aes.secret.key}")
    private String secretKey;

    @Autowired
    private EncryptedCustomerPayloadRepository payloadRepository;

    private final ObjectMapper objectMapper = new ObjectMapper();

    @Override
    public void doFilter(ServletRequest req, ServletResponse res, FilterChain chain)
            throws IOException, ServletException {

        HttpServletRequest httpRequest = (HttpServletRequest) req;
        HttpServletResponse httpResponse = (HttpServletResponse) res;

        if (!shouldDecrypt(httpRequest)) {
            chain.doFilter(req, res);
            return;
        }

        try {
            String encryptedBody = StreamUtils.copyToString(
                    httpRequest.getInputStream(), StandardCharsets.UTF_8);

            if (encryptedBody == null || encryptedBody.isBlank()) {
                chain.doFilter(req, res);
                return;
            }

            SecretKey key = AesGcmUtil.getKeyFromString(secretKey);

            // decrypt
            String decryptedJson = AesGcmUtil.decrypt(encryptedBody, key);
            CustomerDto customerDto = objectMapper.readValue(decryptedJson, CustomerDto.class);

            // validate
            List<String> errors = CustomerValidation.validate(customerDto);
            if (!errors.isEmpty()) {
                sendValidationError(httpResponse, errors);
                return;
            }

            // re-encrypt with fresh IV and save
            String reEncryptedPayload = AesGcmUtil.encrypt(decryptedJson, key);

            EncryptedCustomerPayload entity = new EncryptedCustomerPayload();
            entity.setEncryptedData(reEncryptedPayload);
            entity.setSourceService("order-service");
            entity.setCreatedAt(LocalDateTime.now());
            payloadRepository.save(entity);

            // forward plain data to controller
            DecryptedRequestWrapper wrappedRequest =
                    new DecryptedRequestWrapper(httpRequest, decryptedJson.getBytes(StandardCharsets.UTF_8));

            chain.doFilter(wrappedRequest, res);

        } catch (Exception e) {
            httpResponse.sendError(400, "Invalid encrypted payload: " + e.getMessage());
        }
    }

    private void sendValidationError(HttpServletResponse response, List<String> errors) throws IOException {
        response.setStatus(422);
        response.setContentType("application/json");
        PrintWriter writer = response.getWriter();
        writer.write(objectMapper.writeValueAsString(new ValidationErrorResponse("Validation failed", errors)));
        writer.flush();
    }

    private boolean shouldDecrypt(HttpServletRequest request) {
        String encryptedHeader = request.getHeader("X-Encrypted");
        String path = request.getRequestURI();
        return "true".equalsIgnoreCase(encryptedHeader) && path.startsWith("/api/secure");
    }

    static class ValidationErrorResponse {
        public String message;
        public List<String> errors;
        public ValidationErrorResponse(String message, List<String> errors) {
            this.message = message;
            this.errors = errors;
        }
    }
}

FLOW
====================================================================

1. order-service encrypts CustomerDto -> sends to driver-assignment-service
2. Filter decrypts body -> gets plain CustomerDto
3. Filter validates fields; invalid -> 422 response, controller not called
4. Valid -> filter re-encrypts (fresh IV) and saves to DB
5. Filter forwards decrypted plain JSON to controller
6. Controller runs business logic as normal


====================================================================
WHY RE-ENCRYPT BEFORE DB SAVE
====================================================================

- Never save plain PII -> DB breach exposes customer data directly
- Never reuse caller's ciphertext -> if that copy leaks in transit 
  logs, DB copy would be compromised too
- Fresh IV keeps network copy and DB copy cryptographically independent

INTERVIEW ONE-LINERS

Q: Why validate after decrypt, not before?
A: Ciphertext has no readable fields; validation needs plaintext.

Q: Why re-encrypt instead of storing caller's ciphertext?
A: Fresh IV avoids ciphertext reuse across transit and storage layers.

Q: Why 422 instead of 400 for validation errors?
A: 400 = malformed/undecryptable request. 422 = well-formed but 
   invalid business data.

kafka down what steps follow ?

Kafka down = messages frozen. Your app is trying to produce or consume events, but nothing moves.
The recovery path goes like this:

Check broker health first — are the Kafka processes actually running? This is your fork: if brokers look fine, the issue is elsewhere (logs, network, topic config). If brokers are down, go straight to fixing them.
Check logs and network — look for crash reasons, OOM errors, disk issues, or network partitions. Also check whether topic partitions still have in-sync replicas (ISR). A partition with no in-sync replicas can't accept writes.
Restart the failed broker — once you understand why it died, bring it back. Kafka will automatically re-sync replicas once the broker rejoins the cluster.
Add retry/fallback in your app — even after recovery, your app may have dropped messages during the outage. Use a retry queue, dead-letter topic, or a local buffer so messages don't vanish silently next time.

if db down what do?

Disk full — the DB can't write WAL logs or data files. Free space by cleaning old logs or expanding the volume, then restart.
Out of memory — increase the instance size or tune the buffer pool (e.g. innodb_buffer_pool_size for MySQL, shared_buffers for Postgres).
Bad config — a config file edit broke startup. Check my.cnf or postgresql.conf for typos.

If it's running but unreachable:

Port blocked — firewall or security group is blocking 5432/3306. Check network rules.
Connection pool exhausted — too many app threads holding connections. Increase max_connections or tune your pool size.

How to diagnose:

Read the DB error log first — it almost always tells you exactly why it failed
Check CPU, memory, and disk IOPS — spikes point to resource exhaustion
From the app server, run telnet db-host 5432 or mysql -h db-host to test raw connectivity




Restart after fixing the root cause
Failover to a replica if you have a standby — promote it to primary
Restore from backup if data is corrupted

if redis down?

Default behavior: if Redis is down, @Cacheable/@CachePut/@CacheEvict throw exceptions that propagate and break the method call (it never executes).
Fix: implement a custom CacheErrorHandler and register it via CachingConfigurer — logs the failure and lets the method run normally, skipping the cache.

Optional: wrap direct RedisTemplate calls in a Resilience4j circuit breaker so repeated failures fail fast instead of retrying against a dead Redis.
Enable management.health.redis.enabled=true (Actuator) to detect Redis outages via health checks.






discuss your appproach optimizing database interactions in your projects   

Indexing: add indexes on frequently filtered/joined/sorted columns; avoid over-indexing (slows writes); use composite indexes matching query patterns.
Query optimization: avoid SELECT *, fetch only needed columns; use EXPLAIN ANALYZE to spot full table scans or bad joins.
N+1 problem: biggest one in Spring/JPA — fixed using JOIN FETCH, @EntityGraph, or batch fetching instead of lazy-loading in loops.
Pagination: use Pageable/keyset pagination instead of loading large result sets into memory.
Caching: cache read-heavy, rarely-changing data (Redis/Caffeine) to cut DB round trips.
Connection pooling: tune HikariCP pool size based on load testing, not defaults.
Batch operations: use batch inserts/updates (spring.jpa.properties.hibernate.jdbc.batch_size) instead of row-by-row writes.
Transaction scope: keep transactions short and narrow to reduce lock contention.
Read/write separation: route read-heavy traffic to read replicas where applicable.
Denormalization: selectively denormalize for reporting/read-heavy paths where joins are too costly.
Monitoring: track slow queries (pg_stat_statements, slow query logs) and review them periodically.
Schema design: choose proper data types, avoid unnecessary nullable columns, normalize appropriately for the use case.






SQL Injection prevention:

Use JPA/Hibernate with parameterized queries (@Query with named params) — never string-concatenate SQL.
If using JdbcTemplate, always use ? placeholders, never raw string building.
Avoid createNativeQuery with concatenated input; use bind parameters even there.
Validate/sanitize input with @Valid + Bean Validation annotations (@Pattern, @Size, etc.).



XSS prevention:

Escape/encode output when rendering user input (Thymeleaf auto-escapes by default — avoid th:utext for untrusted data).
For REST APIs returning JSON, ensure proper Content-Type: application/json (browsers won't execute JSON as HTML/script).
Sanitize rich text input (e.g., comments, descriptions) using a library like OWASP Java HTML Sanitizer if HTML must be allowed.
Set security headers via Content-Security-Policy, X-XSS-Protection, X-Content-Type-Options: nosniff (can configure via Spring Security's HeaderWriter).

import lombok.extern.slf4j.Slf4j;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.context.request.WebRequest;

import java.nio.file.AccessDeniedException;
import java.time.LocalDateTime;
import java.util.HashMap;
import java.util.Map;

@RestControllerAdvice
@Slf4j
public class RealTimeGlobalExceptionHandler {

    // 1. ALL CUSTOM BUSINESS EXCEPTIONS (Clean & Centralized)
    @ExceptionHandler(BaseBusinessException.class)
    public ResponseEntity<ApiErrorResponse> handleBusinessExceptions(BaseBusinessException ex, WebRequest request) {
        log.warn("Business rule violation: {} - Code: {}", ex.getMessage(), ex.getErrorCode());
        
        ApiErrorResponse response = buildErrorResponse(ex.getHttpStatus(), ex.getMessage(), ex.getErrorCode(), request);
        return new ResponseEntity<>(response, ex.getHttpStatus());
    }

    // 2. SPRING VALIDATION EXCEPTIONS (When DTO validation e.g., @NotBlank fails)
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiErrorResponse> handleValidationExceptions(MethodArgumentNotValidException ex, WebRequest request) {
        log.warn("Validation failed for request: {}", request.getDescription(false));
        
        Map<String, String> errors = new HashMap<>();
        ex.getBindingResult().getAllErrors().forEach((error) -> {
            String fieldName = ((FieldError) error).getField();
            String errorMessage = error.getDefaultMessage();
            errors.put(fieldName, errorMessage);
        });

        ApiErrorResponse response = ApiErrorResponse.builder()
                .timestamp(LocalDateTime.now())
                .status(HttpStatus.BAD_REQUEST.value())
                .error(HttpStatus.BAD_REQUEST.getReasonPhrase())
                .customErrorCode("VALIDATION_ERROR")
                .message("Input payload validation failed")
                .validationErrors(errors)
                .path(request.getDescription(false).replace("uri=", ""))
                .build();

        return new ResponseEntity<>(response, HttpStatus.BAD_REQUEST);
    }

    // 3. DATABASE CONSTRAINT VIOLATIONS (Catching unique constraint drops before they leak)
    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ApiErrorResponse> handleDatabaseConstraints(DataIntegrityViolationException ex, WebRequest request) {
        log.error("Database constraint violation detected!", ex);
        
        ApiErrorResponse response = buildErrorResponse(
                HttpStatus.CONFLICT, 
                "Database error: Data integrity violation or duplicate key.", 
                "DB_CONSTRAINT_ERROR", 
                request
        );
        return new ResponseEntity<>(response, HttpStatus.CONFLICT);
    }

    // 4. SECURITY / AUTHORIZATION EXCEPTIONS
    @ExceptionHandler(AccessDeniedException.class)
    public ResponseEntity<ApiErrorResponse> handleAccessDenied(AccessDeniedException ex, WebRequest request) {
        log.warn("Unauthorized access attempt on path: {}", request.getDescription(false));
        
        ApiErrorResponse response = buildErrorResponse(
                HttpStatus.FORBIDDEN, 
                "You do not have permission to access this resource.", 
                "AUTH_FORBIDDEN", 
                request
        );
        return new ResponseEntity<>(response, HttpStatus.FORBIDDEN);
    }

    // 5. THE ULTIMATE CATCH-ALL (Prevents raw stack traces from exposing system internals)
    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiErrorResponse> handleAllUncaughtExceptions(Exception ex, WebRequest request) {
        // Critical: Log the full stack trace internally so developers can fix it via logs
        log.error("CRITICAL UNCAUGHT EXCEPTION: ", ex);
        
        ApiErrorResponse response = buildErrorResponse(
                HttpStatus.INTERNAL_SERVER_ERROR, 
                "A critical internal server error occurred. Please contact support.", 
                "SYS_INTERNAL_ERROR", 
                request
        );
        return new ResponseEntity<>(response, HttpStatus.INTERNAL_SERVER_ERROR);
    }

    // Helper method to keep code DRY
    private ApiErrorResponse buildErrorResponse(HttpStatus status, String message, String customCode, WebRequest request) {
        return ApiErrorResponse.builder()
                .timestamp(LocalDateTime.now())
                .status(status.value())
                .error(status.getReasonPhrase())
                .customErrorCode(customCode)
                .message(message)
                .path(request.getDescription(false).replace("uri=", ""))
                .build();
    }
}




Never return ex.getMessage() on raw Exception.class: If a database connection fails, returning ex.getMessage() could leak database URLs, usernames, or table schemas to a malicious client.

What is the difference between @ControllerAdvice and @RestControllerAdvice?
What they are testing: Your understanding of Spring MVC basics.

The Answer: @RestControllerAdvice is a convenience annotation that combines @ControllerAdvice and @ResponseBody.

With @ControllerAdvice, methods return a view name (HTML page) by default unless you add @ResponseBody to every method.

With @RestControllerAdvice, Spring automatically serializes the returned object (like your ErrorResponse) directly into JSON/XML in the HTTP response body.



How do you handle exceptions that occur before reaching the Controller (e.g., in a Filter or Spring Security)?
What they are testing: Understanding the Spring request lifecycle.

The Answer: @RestControllerAdvice only catches exceptions thrown within the DispatcherServlet's scope (Controller, Service, Repository layers). It cannot catch exceptions thrown inside Servlet Filters or Spring Security filters because filters execute before the request ever reaches the dispatcher. To handle filter exceptions, you must either handle them inside the filter directly or configure an AuthenticationEntryPoint / AccessDeniedHandler in your Spring Security configuration.


If two @ExceptionHandler methods match the same exception hierarchy (e.g., one catches IOException and another catches Exception), which one will run?
What they are testing: Spring's exception matching algorithm.

The Answer: Spring always looks for the most specific match first. If an IOException is thrown, the handler explicitly annotated for IOException will execute. The generic Exception handler will only pick it up if a more specific handler doesn't exist.


How do you maintain the order of execution if you have multiple @RestControllerAdvice classes?
What they are testing: Microservices or modular architecture design.

The Answer: If you have multiple advice classes (e.g., one in a shared library module and one in your main application), you can control their priority using the @Order annotation or by implementing the Ordered interface. Lower values have higher priority (e.g., @Order(1) runs before @Order(2)).


f a controller method has a try-catch block, will @RestControllerAdvice still catch that exception?
What they are testing: Your understanding of Java basics and Spring's interceptor lifecycle.
The Answer: No. Spring's AOP (Aspect-Oriented Programming) proxy intercepts exceptions after they escape the controller layer. If you catch and handle the exception inside the controller method, it never bubbles up to Spring, so the global exception handler will never see it.

        studentSet.add(alex2); 
    }
}

What Java does behind the scenes on studentSet.add(alex2):
Java calls alex2.hashCode(): It gets the number 101. Java instantly jumps straight to the "Bucket 101" in memory. It doesn't look anywhere else.

Java looks inside Bucket 101: It sees alex1 is already sitting there.

Java calls alex1.equals(alex2): It checks if the names and IDs match perfectly. They do!

Result: Java says, "Ah, this is a duplicate!" and rejects alex2.

Without hashCode(), Java would have to scan the entire memory to find alex1. Without equals(), Java wouldn't know for sure if they were actually the same person.

Database Patterns (How to manage data)
Data is arguably the hardest part of microservices. Sharing a database creates tight coupling, but separating them creates consistency issues.

Database per Service: The golden rule of microservices. Each service owns its database, and other services can only access that data through the owning service's API. This ensures loose coupling.

Saga Pattern: Because you can't easily use traditional ACID transactions across multiple databases, a Saga manages distributed transactions as a sequence of local transactions. If one step fails, the Saga executes compensating transactions to undo the preceding steps.


Strangler Fig Pattern: Used for migrating a legacy monolith to microservices. You incrementally extract functionalities into new microservices and route traffic to them, gradually "strangling" the old monolith until it can be retired.



Integration & Routing Patterns (How they communicate)
Clients (web, mobile) need a cohesive way to talk to dozens of fragmented backend services.

API Gateway: Acts as a single entry point into the system for all clients. It handles request routing, composition, and cross-cutting concerns like authentication, rate limiting, and SSL termination.

BFF (Backend for Frontend): A variation of the API Gateway where you create a separate API Gateway for each type of UI (e.g., one for the mobile app, one for the web app). This prevents a single gateway from becoming a bottleneck and allows each client to get exactly the data shape it needs.









502 Bad Gateway error



The Upstream Service is Down or Unreachable
This is the most common cause. The gateway is trying to route traffic to a service that is not there.

Application Crash: The Spring Boot application has crashed (e.g., due to an OutOfMemoryError or fatal exception) and the process has terminated.

Deployment Downtime: During a CI/CD pipeline deployment, if zero-downtime deployment (like blue-green or rolling updates) is not configured correctly, requests hitting the gateway while the new instance is starting up will result in a 502.

Failed Health Checks: The container orchestration system (like Kubernetes) or load balancer marks the Spring Boot pod/instance as unhealthy and terminates it, but the gateway hasn't updated its routing table yet.

2. Service Discovery (Registry) Sync Issues
If you are using a service registry like Netflix Eureka, HashiCorp Consul, or Kubernetes CoreDNS, the gateway relies on this registry to know where services live.

Stale Registry Data: When a Spring Boot instance scales down or crashes, it takes a few seconds (or minutes, depending on your cache configurations) for the registry to remove it. During this window, the API Gateway still routes traffic to the dead IP address.

Missing Instances: The service is running, but it failed to register itself with the discovery server due to misconfiguration (e.g., wrong eureka.client.serviceUrl.defaultZone).

3. Resource Exhaustion (Tomcat/Undertow)
Even if the Spring Boot application is technically "running," it might be in a state where it cannot accept or process new TCP connections.

Thread Pool Exhaustion: Spring Boot’s embedded server (Tomcat by default) has a maximum number of worker threads (default server.tomcat.threads.max=200). If all threads are busy (often due to blocked database queries or slow external API calls), the server will reject new connections, causing the gateway to throw a 502.

Connection Queue Full: If the thread pool is full, Tomcat places requests in a queue (the accept-count, default 100). If this queue also fills up, subsequent connections are immediately refused by the OS.



Firewall/Security Groups:


Check the Gateway Logs First: Find the exact IP and port the gateway was trying to reach when the 502 occurred.

Verify the IP: Check your Service Registry (Eureka/Consul) or Kubernetes Endpoints. Does that IP actually exist right now?

Check Upstream Logs: Look at the logs of the specific Spring Boot instance. If there are no logs for the request, the request never reached it (pointing to Network/Registry issues). If there are OutOfMemory or database timeout logs, you have a resource exhaustion issue.

External API Latency: Your service relies on a third-party API (like a slow payment gateway or external logistics tracker) that is dragging its feet.

Thread Locks/Deadlocks: The Tomcat worker threads in the Spring Boot app are completely stuck waiting on a synchronized block or a locked database row, preventing them from writing an HTTP response.

To implement Client-Side Load Balancing in modern Spring Boot, we use Spring Cloud LoadBalancer. (Note: Netflix Ribbon is completely deprecated).

While a standard load balancer (like AWS ALB or Nginx) sits in the middle of your network, a client-side load balancer lives directly inside your Spring Boot application. It looks at your Service Registry (like Eureka), gets a list of all available IPs for a target service, and mathematically decides which one to call.


Step 1: Add the Dependency
Include the Spring Cloud LoadBalancer starter in your pom.xml. (Assuming you already have a service registry client like Eureka installed).

XML
<dependency>
    <groupId>org.springframework.cloud</groupId>
    <artifactId>spring-cloud-starter-loadbalancer</artifactId>
</dependency>


Step 2: Apply the Annotation (The Magic Step)
You don't need complex configurations to get started. You simply add the @LoadBalanced annotation to your RestTemplate or WebClient bean.

Java

import org.springframework.cloud.client.loadbalancer.LoadBalanced;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.web.client.RestTemplate;

@Configuration
public class AppConfig {

    @Bean
    @LoadBalanced // This single annotation intercepts calls and distributes them!
    public RestTemplate restTemplate() {
        return new RestTemplate();
    }
}



Step 3: Call by Service Name (Not IP)
Because the RestTemplate is now load-balanced, you stop using hardcoded IP addresses. Instead, you use the exact registered name of the downstream microservice.

Java
@Service
public class BiddingService {

    private final RestTemplate restTemplate;

    public BiddingService(RestTemplate restTemplate) {
        this.restTemplate = restTemplate;
    }

    public BidResponse submitBid(BidRequestDTO request) {
        // Notice the URL: http://{SERVICE-NAME}/api/...
        // The LoadBalancer will automatically swap "bidding-service" with a real IP and Port!
        return restTemplate.postForObject("http://bidding-service/api/bids", request, BidResponse.class);
    }

Pattern 1: Service Discovery + Client-Side Load Balancing (Spring Cloud Native)
If your microservices are deployed across EC2 instances or container groups where instances scale up and down frequently, this is the most common Spring Boot approach.

How it works: You deploy a private Service Registry (like Netflix Eureka or HashiCorp Consul) inside your private VPC. When Service A wants to call Service B, it asks the registry for Service B's active internal IP addresses. Using Spring Cloud LoadBalancer, Service A picks an IP and calls Service B directly.

To set a Time-To-Live (TTL), you need to create a dedicated configuration class that overrides Spring's default caching behavior.


port org.springframework.data.redis.cache.RedisCacheManager;
import org.springframework.data.redis.connection.RedisConnectionFactory;

import java.time.Duration;
import java.util.HashMap;
import java.util.Map;

@Configuration
@EnableCaching
public class RedisIndividualTTLConfig {

    @Bean
    public RedisCacheManager cacheManager(RedisConnectionFactory connectionFactory) {
        
        // 1. Define the Global Default (Fallback)
        RedisCacheConfiguration defaultCfg = RedisCacheConfiguration.defaultCacheConfig()
            .entryTtl(Duration.ofMinutes(10));

        // 2. Define specific TTLs for specific caches
        Map<String, RedisCacheConfiguration> cacheConfigs = new HashMap<>();
        
        // Products cache expires in 5 minutes
        cacheConfigs.put("products", RedisCacheConfiguration.defaultCacheConfig()
            .entryTtl(Duration.ofMinutes(5)));
            
        // Zipcodes cache expires in 24 hours
        cacheConfigs.put("zipcodes", RedisCacheConfiguration.defaultCacheConfig()
            .entryTtl(Duration.ofHours(24)));

        // 3. Build the Manager with initial configs
        return RedisCacheManager.builder(connectionFactory)
            .cacheDefaults(defaultCfg)
            .withInitialCacheConfigurations(cacheConfigs) // <--- This applies the map
            .build();
    }





The Fallback: If you use an annotation like @Cacheable("orders") and it isn't found in your map, Spring gracefully falls back to the defaultCfg (10 minutes).
